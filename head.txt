
#include <iostream>
using namespace std;

struct student {
    string name;
    student* next;
};

int main() {

    // Initially head points to nullptr
    student* head = nullptr;


    // ========== FIRST NODE START ==========

    student* first = new student();

    first->name = "Fasih";

    first->next = head;

    head = first;  // Head now points to first node

    // ========== FIRST NODE END ==========


    // ========== SECOND NODE START ==========

    student* second = new student();

    second->name = "Fasih";

    second->next = head;

    head = second; // Head now points to second node

    // ========== SECOND NODE END ==========


    // ========== THIRD NODE START ==========

    student* third = new student();

    third->name = "Ammar";

    third->next = head;

    head = third; // Head now points to third node

    // ========== THIRD NODE END ==========


    // ========== FOURTH NODE START ==========

    student* fourth = new student();

    fourth->name = "Mehdi";

    fourth->next = head;

    head = fourth; // Head now points to fourth node

    // ========== FOURTH NODE END ==========


    // ========== FIFTH NODE START ==========

    student* fifth = new student();

    fifth->name = "Abrar";

    fifth->next = head;

    head = fifth; // Head now points to fifth node

    // ========== FIFTH NODE END ==========


    // ========== TRAVERSAL START ==========

    student* temp = head;

    int count = 0;

    while (temp != nullptr) {

        cout << "Student Name: " << temp->name << endl;

        cout << "Node Address: " << temp << endl;

        cout << "Next Address: " << temp->next << endl;

        cout << "------------------------" << endl;

        count++;

        temp = temp->next;
    }

    // ========== TRAVERSAL END ==========


    // Display total nodes

    cout << "Total Nodes: " << count << endl;


    // Last node's next address is nullptr

    cout << "Last Node Next Address: " << temp << endl;


    return 0;
}
